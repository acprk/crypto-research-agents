\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath}
\usepackage{crypto-tikz}
\begin{document}
\begin{tikzpicture}[node distance=5mm]
  \node (in)  {$\mathrm{ct}$};
  \node[box, right=of in] (s1) {ModRaise};
  \node[box, right=of s1] (s2) {CoeffToSlot};
  \node[ours, right=of s2] (s3) {EvalMod\\[-1pt]{\footnotesize (new)}};
